// Scheduled Task component usage C++/VC++ samples
// #1
// example creates new scheduled task called "myTask"
// task should start notepad on May 5, 2001 10:20AM once

// use something like this
#import "D:\\Path_to_my_SchedTask\\SchedTask.tlb"   // in all these examples below

CoInitialize(NULL);

try
{
    SCHEDTASKLib::IetScheduledTaskPtr pSTask(__uuidof(SCHEDTASKLib::etScheduledTask));
    pSTask->Program = "notepad.exe";
    pSTask->StartingDir = "C:\\Winnt\\system32";

    pSTask->TriggerType = 0;    // once
    pSTask->StartYear = 2001;
    pSTask->StartMonth = 5;
    pSTask->StartDay = 5;
    pSTask->StartHour = 10;
    pSTask->StartMinute = 20;

    pSTask->DontStartIfOnBatteries = FALSE
    pSTask->KillIfOnBatteries = FALSE

    pSTask->Username = "TaskRun"
    pSTask->Password = "taskrun"    // password is case sensitive
    pSTask->Creator = "ME"

    pSTask->CreateTask("myTask", TRUE);         // calls AddTrigger inside
   
}
catch(const _com_error &err)
{
    // retrieve rich error information here
}

CoUninitialize();

// #6
// example enumerates all existing tasks
// each task has extension ".job" . If you want to compare task you have
// created with existing task - drop/trim the extension .

#include "atlconv.h" // cause we use some ATL macros for conversion

USES_CONVERSION;
CoInitialize(NULL);

try
{
    SCHEDTASKLib::IetScheduledTaskPtr pSTask(__uuidof(SCHEDTASKLib::etScheduledTask));

    char szTask[120];
    pSTask->Enum();

    for (int i=1; i < pSTask->TaskCount; i++)
    {
        pSTask->TaskItem = i;
        strncpy(szTask, OLE2T(pSTask->TaskName), 120);
    }   

}
catch(const _com_error &err)
{
    // retrieve rich error information here
    _ASSERT(0);
}

CoUninitialize();
