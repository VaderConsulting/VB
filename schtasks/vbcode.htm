    ' Scheduled Task component usage VB samples
    ' check "SchedTask 1.0 Type Library" in Tools|References

    ' #1
    ' example creates new scheduled task called "myTask"
    ' task should start notepad on May 5, 2001 10:20AM once

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.Parameters = "myinputfile.txt"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 0 ' once

    obj.StartYear = 2001
    obj.StartMonth = 5
    obj.StartDay = 5
    obj.StartHour = 10
    obj.StartMinute = 20

    obj.DontStartIfOnBatteries = False   ' default is True
    obj.KillIfOnBatteries = False             ' default is True

    obj.Username = "TaskRun"
    obj.Password = "taskrun"    ' password is case sensitive
    obj.Creator = "Me"             ' default user if omited

    obj.CreateTask "myTask", True ' calls AddTrigger inside
    obj.EditTask "myTask"                 ' give user dialog to edit

    ; #2
    ; example creates new scheduled task called "myTask2"
    ; task should start notepad.exe every 15 minutes during 2 hours (120min.)
    ; starting May 5, 2001 10:20AM daily
   
    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 1        ' daily

    obj.StartYear = 2001
    obj.StartMonth = 5
    obj.StartDay = 5
    obj.StartHour = 10
    obj.StartMinute = 20

    obj.MinutesInterval = 15
    obj.MinutesDuration = 120

    obj.CreateTask "myTask2", TRUE    ' calls AddTrigger inside


    ' #3
    ' example creates new scheduled task called "myTask3"
    ' task should start notepad.exe at logon

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 7        ' at logon
   
    obj.CreateTask("myTask3", TRUE)         ' calls AddTrigger inside


    ' #4
    ' example creates new scheduled task called "myTask4"
    ' task should start notepad.exe every sunday of every week
    ' starting May 5, 2001 10:20AM
    ' If task named "myTask4" already exists it will be overwritten

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 2        ' weekly
    obj.DayOfTheWeek = 1    ' every sunday

    obj.StartYear = 2001
    obj.StartMonth = 5
    obj.StartDay = 5
    obj.StartHour = 10
    obj.StartMinute = 20
   
    obj.CreateTask "myTask4", FALSE         ' calls AddTrigger inside


    ' #5
    ' example creates new scheduled task called "myTask5"
    ' task should start al 10:40AM on day 2 of every month,
    ' starting 05/04/2000 and ending 05/10/2001

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 3        ' monthly
    obj.DayOfTheWeek = 2    ' every second day of the month

    obj.StartYear = 2001
    obj.StartMonth = 5
    obj.StartDay = 4
    obj.StartHour = 10
    obj.StartMinute = 20

    pSTask->EndYear = 2001
    pSTask->EndMonth = 5
    pSTask->EndDay = 10
   
    obj.CreateTask "myTask5", FALSE         ' calls AddTrigger inside


// #6
// example enumerates all existing tasks
// each task has extension ".job" . If you want to compare task you have
// created with existing task - drop/trim the extension

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Enum   
    i = 1

    while (i < obj.TaskCount)   ' loop thru all tasks
   
        obj.TaskItem = i             ' setup task index
        Str = obj.TaskName        ' get task name
        MsgBox Str
        i = i + 1

    wend


// #7
//    Run on specific months only
//

    Dim obj
    Set obj = CreateObject("SCHEDTASK.etScheduledTask")

    obj.Program = "notepad.exe"
    obj.StartingDir = "C:\Winnt\system32"

    obj.TriggerType = 3 ' monthly

    obj.StartYear = 2000
    obj.StartMonth = 7
    obj.StartDay = 28
    obj.StartHour = 18
    obj.StartMinute = 53

    obj.AddMonth 1        ' run January and February
    obj.AddMonth 2

    obj.DayoftheMonth = 15

    obj.DontStartIfOnBatteries = False    ' default is True
    obj.KillIfOnBatteries = False         ' default is True

    obj.CreateTask "myTask1", False ' calls AddTrigger inside

// #11
// get enabled/disabled flag for task
//
   Dim obj, s
   Set obj = CreateObject("SCHEDTASK.etScheduledTask")
   obj.ActivateTask "Task2" 
   If (obj.Disabled = 1) Then
         MsgBox "disabled" 
   End If


// #12
// get trigger string (textual trigger description)
//
   Dim obj, s
   Set obj = CreateObject("SCHEDTASK.etScheduledTask")
   obj.ActivateTask "Task2"
   obj.GetTrigger 0
   s = obj.TriggerString
   MsgBox s
